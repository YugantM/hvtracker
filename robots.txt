# /login?next= chains self-nest (421 crawled URLs, up to 7 levels deep in

Disallow: /login

# Parameterized compare-hub fallback (/compare/?a=&b=) 200s but can never
# index (canonical points to the static pair); already X-Robots noindex,
# this stops the crawl outright. Static /compare/<a>-vs-<b>/ pages stay allowed.
Disallow: /compare/?a=
# Badge SVGs are embedded on every agent page (2 per page, ~2,656 URLs) and
# regenerate each refresh, so their Last-Modified is always new and Googlebot
# re-downloads instead of getting a 304 — ~35% of crawl budget. GitHub's camo
# proxy fetches badges server-side and ignores robots.txt, so the badge program
# and badge_fetches KPI are unaffected.
Disallow: /badge/
